5. Technical SQL Challenges & Live Problems (Round 3 & Funnel Patterns)

R3 Problem 1: State Policy Distribution & Data Hygiene

Pushan's challenge: Group active policies by property state, calculate total active policies, and calculate the percentage of total active policies using window functions.

Window Function Ratio 
SELECT
    pr.state AS property_state,
    COUNT(DISTINCT p.policy_id) AS active_policy_count,
    ROUND(
        COUNT(DISTINCT p.policy_id) * 100.0 / 
        SUM(COUNT(DISTINCT p.policy_id)) OVER (), 
        2
    ) AS pct_of_total_policies
FROM policies p
JOIN properties pr ON p.property_id = pr.property_id
WHERE p.policy_status = 'active'
GROUP BY pr.state
ORDER BY active_policy_count DESC;

R3 Problem 2: First-Claim Loss by Tier (ROW_NUMBER)

Identify each policy's earliest filed claim, aggregate total losses by customer risk tier, and explain the loss ratio denominator.

ROW_NUMBER() Earliest Claim 
WITH ranked_claims AS (
    SELECT
        c.claim_id,
        c.policy_id,
        c.amount_paid,
        c.claim_date,
        ROW_NUMBER() OVER (
            PARTITION BY c.policy_id 
            ORDER BY c.claim_date ASC
        ) AS rn
    FROM claims c
)
SELECT
    cu.customer_tier,
    COUNT(DISTINCT rc.policy_id) AS policies_with_claims,
    SUM(rc.amount_paid) AS total_first_claim_losses
FROM ranked_claims rc
JOIN policies p ON rc.policy_id = p.policy_id
JOIN properties pr ON p.property_id = pr.property_id
JOIN customers cu ON pr.customer_id = cu.customer_id
WHERE rc.rn = 1
GROUP BY cu.customer_tier
ORDER BY total_first_claim_losses DESC;

R3 Problem 3: Junior Analyst Query Bug Fix

Fix junior analyst query that had a global WHERE clause on LEFT JOIN (eliminating zero-claim policies) and duplicate premium sums due to claim fan-out.

Pre-Aggregated CTE Pattern 
WITH policy_claims AS (
    SELECT
        policy_id,
        SUM(amount_paid) AS total_loss_paid,
        COUNT(claim_id) AS total_claims
    FROM claims
    WHERE claim_date >= '2026-01-01'
    GROUP BY policy_id
)
SELECT
    p.policy_id,
    p.annual_premium,
    COALESCE(pc.total_loss_paid, 0) AS total_loss_paid,
    COALESCE(pc.total_claims, 0) AS total_claims
FROM policies p
LEFT JOIN policy_claims pc ON p.policy_id = pc.policy_id
WHERE p.policy_status = 'active';

Funnel Pattern 1: Quote-to-Bind Funnel (30-Day Window)

Measure quotes, binds within 30 days, bind rate, and total premium by channel via LEFT JOIN in ON clause.

Date Window in ON Clause 
SELECT
    q.channel,
    COUNT(DISTINCT q.quote_id) AS total_quotes,
    COUNT(DISTINCT p.policy_id) AS total_binds,
    COUNT(DISTINCT p.policy_id) * 1.0 / COUNT(DISTINCT q.quote_id) AS bind_rate,
    SUM(p.annual_premium) AS total_premium
FROM quotes q
LEFT JOIN policies p
    ON q.quote_id = p.quote_id
   AND p.bound_ts >= q.quote_ts
   AND p.bound_ts < DATEADD(day, 30, q.quote_ts)
WHERE q.quote_ts >= '2026-07-01'
  AND q.quote_ts < '2026-10-01'
GROUP BY q.channel;

Funnel Pattern 2: Last-Touch Ad Attribution (ROW_NUMBER)

Attribute each quote to the single most recent ad click in the 14 days prior to quote.

Most Recent Click Within 14 Days 
WITH ranked_clicks AS (
    SELECT
        q.quote_id, q.user_id, q.quote_ts,
        c.channel, c.campaign_id, c.click_ts,
        ROW_NUMBER() OVER (
            PARTITION BY q.quote_id
            ORDER BY c.click_ts DESC
        ) AS rn
    FROM quotes q
    LEFT JOIN ad_clicks c
        ON q.user_id = c.user_id
       AND c.click_ts <= q.quote_ts
       AND c.click_ts >= DATEADD(day, -14, q.quote_ts)
)
SELECT quote_id, user_id, quote_ts, channel AS last_touch_channel,
       campaign_id AS last_touch_campaign, click_ts AS last_touch_click_ts
FROM ranked_clicks
WHERE rn = 1;

Funnel Pattern 3: Spend & Binds at Matching Grain (No Fan-Out)

Aggregate spend to month/channel in CTE 1, aggregate policies in CTE 2, join 1-to-1 to calculate CAC & ROAS.

CTE Grain Alignment 
WITH monthly_spend AS (
    SELECT DATE_TRUNC('month', spend_date) AS cohort_month, channel, SUM(spend) AS total_spend
    FROM daily_ad_spend GROUP BY 1, 2
),
monthly_binds AS (
    SELECT DATE_TRUNC('month', p.bound_date) AS cohort_month, s.channel,
           COUNT(DISTINCT p.policy_id) AS total_binds, SUM(p.annual_premium) AS total_premium
    FROM policies p
    JOIN (SELECT DISTINCT campaign_id, channel FROM daily_ad_spend) s
        ON p.campaign_id = s.campaign_id
    GROUP BY 1, 2
)
SELECT ms.cohort_month, ms.channel, ms.total_spend, mb.total_binds,
       ms.total_spend / NULLIF(mb.total_binds, 0) AS cost_per_bind,
       mb.total_premium / NULLIF(ms.total_spend, 0) AS roas
FROM monthly_spend ms
LEFT JOIN monthly_binds mb
    ON ms.cohort_month = mb.cohort_month AND ms.channel = mb.channel
ORDER BY ms.cohort_month DESC, ms.total_spend DESC;

Healthcare Funnel & Visit Conversion

Patient Lead -> Booked -> Completed Visit 
WITH first_visit AS (
    SELECT
        patient_id,
        MIN(CASE WHEN status = 'completed' THEN visit_date END) AS first_completed_dt,
        COUNT(CASE WHEN status = 'completed' THEN 1 END) AS completed_visits
    FROM appointments
    GROUP BY patient_id
)
SELECT
    p.acquisition_channel,
    COUNT(DISTINCT p.patient_id) AS total_leads,
    COUNT(DISTINCT fv.first_completed_dt) AS converted_patients,
    ROUND(100.0 * COUNT(DISTINCT fv.first_completed_dt) / COUNT(DISTINCT p.patient_id), 1) AS conv_pct
FROM patients p
LEFT JOIN first_visit fv
    ON p.patient_id = fv.patient_id
GROUP BY p.acquisition_channel;

Cohort Retention & Repeat Visits (LAG / Window)

Days Between Completed Appointments 
WITH ordered_visits AS (
    SELECT
        patient_id,
        visit_id,
        visit_date,
        ROW_NUMBER() OVER (PARTITION BY patient_id ORDER BY visit_date) AS visit_num,
        LAG(visit_date) OVER (PARTITION BY patient_id ORDER BY visit_date) AS prev_visit_date
    FROM appointments
    WHERE status = 'completed'
)
SELECT
    visit_num,
    COUNT(*) AS patient_count,
    AVG(DATEDIFF('day', prev_visit_date, visit_date)) AS avg_days_since_prior
FROM ordered_visits
WHERE prev_visit_date IS NOT NULL
GROUP BY visit_num
ORDER BY visit_num;

Pattern A: Lead -> Completed Visit Funnel

CTE First-Visit Aggregation + Conversion % 
WITH first_visit AS (
    SELECT
        patient_id,
        MIN(CASE WHEN status = 'completed' THEN visit_date END) AS first_completed_dt
    FROM appointments
    GROUP BY patient_id
)
SELECT
    p.acquisition_channel,
    COUNT(DISTINCT p.patient_id) AS total_leads,
    COUNT(DISTINCT fv.first_completed_dt) AS converted_patients,
    ROUND(100.0 * COUNT(DISTINCT fv.first_completed_dt) / NULLIF(COUNT(DISTINCT p.patient_id), 0), 1) AS conv_pct
FROM patients p
LEFT JOIN first_visit fv ON p.patient_id = fv.patient_id
GROUP BY p.acquisition_channel;

Pattern B: Repeat Visit Cadence (LAG)

ROW_NUMBER + LAG Window Function 
WITH ordered_visits AS (
    SELECT
        patient_id,
        visit_date,
        ROW_NUMBER() OVER (PARTITION BY patient_id ORDER BY visit_date) AS visit_num,
        LAG(visit_date) OVER (PARTITION BY patient_id ORDER BY visit_date) AS prev_visit_date
    FROM appointments
    WHERE status = 'completed'
)
SELECT
    visit_num,
    COUNT(*) AS patient_count,
    AVG(DATEDIFF('day', prev_visit_date, visit_date)) AS avg_days_since_prior
FROM ordered_visits
WHERE prev_visit_date IS NOT NULL
GROUP BY visit_num
ORDER BY visit_num;
